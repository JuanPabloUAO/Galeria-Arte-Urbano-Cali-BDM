    DECLARE v_count INT DEFAULT 0;

    SELECT COUNT(*) INTO v_count FROM `artista` WHERE `idArtista` = p_id_artista AND `estadoArtista` = TRUE;
    IF v_count = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El artista seleccionado no existe o está inactivo.';

    IF p_id_obra IS NULL OR TRIM(p_id_obra) = '' THEN
        SET p_nuevo_id_obra = UUID();

        SET p_nuevo_id_obra = p_id_obra;

        public ConexionMysql(
            string servidor = "localhost",
            string baseDatos = "galeria_arte_urbano_db",
            string usuario = "root",
            string clave = "root",
            uint puerto = 3306)
        {
            var builder = new MySqlConnectionStringBuilder
            {
                Server = servidor,
                Database = baseDatos,
                UserID = usuario,
                Password = clave,
                Port = puerto,
                CharacterSet = "utf8mb4"
            };

            _cadenaConexion = builder.ConnectionString;
        }

        public MySqlConnection Abrir()
        {
            _conexion ??= new MySqlConnection(_cadenaConexion);
            if (_conexion.State != ConnectionState.Open)
            {
                _conexion.Open();
            }

            return _conexion;
        }

        public void Cerrar()
        {
            if (_conexion != null && _conexion.State != ConnectionState.Closed)
            {
                _conexion.Close();
            }
        }

        public void Dispose()
        {
            Cerrar();
            _conexion?.Dispose();
            _conexion = null;
        }

        public bool ProbarConexion(out string mensaje)
        {
            try
            {
                using var cn = new MySqlConnection(_cadenaConexion);
                cn.Open();
                mensaje = $"Conectado a MySQL {cn.ServerVersion} (BD: galeria_arte_urbano_db)";
                return true;
            }
            catch (MySqlException ex)
            {
                mensaje = $"Error MySQL ({ex.Number}): {ex.Message}";
                return false;
            }
            catch (Exception ex)
            {
                mensaje = $"Error general: {ex.Message}";
                return false;
            }
        }

        public bool ExecuteQuery(string sql, Dictionary<string, object?> parametros, out string mensajeError)
        {
            bool result = false;
            mensajeError = string.Empty;
            UltimoError = string.Empty;

            try
            {
                using MySqlCommand cmd = new MySqlCommand(sql, Abrir());
                if (parametros != null)
                {
                    foreach (var par in parametros)
                    {
                        cmd.Parameters.AddWithValue(par.Key, par.Value ?? DBNull.Value);
                    }
                }
                int filas = cmd.ExecuteNonQuery();
                result = filas >= 0;
            }
            catch (Exception ex)
            {
                UltimoError = ex.Message;
                mensajeError = ex.Message;
            }
            finally
            {
                Cerrar();
            }
            return result;
        }

        public bool InsertarObra(Obra o, out string mensajeError)
        {
            mensajeError = string.Empty;
            using var conexion = new ConexionMysql();

            try
            {
                using MySqlCommand cmd = new MySqlCommand("sp_registrar_obra", conexion.Abrir());
                cmd.CommandType = CommandType.StoredProcedure;

                string idEntrada = string.IsNullOrWhiteSpace(o.IdObra) ? Guid.NewGuid().ToString() : o.IdObra.Trim();

                cmd.Parameters.AddWithValue("@p_id_obra", idEntrada);
                cmd.Parameters.AddWithValue("@p_nombre_obra", o.NombreObra.Trim());
                cmd.Parameters.AddWithValue("@p_descripcion_obra", string.IsNullOrWhiteSpace(o.DescripcionObra) ? DBNull.Value : o.DescripcionObra.Trim());
                cmd.Parameters.AddWithValue("@p_fecha_creacion", o.FechaCreacion.ToString("yyyy-MM-dd"));
                cmd.Parameters.AddWithValue("@p_estado_conservacion", o.EstadoConservacion.Trim());
                cmd.Parameters.AddWithValue("@p_id_artista", o.IdArtistaFK.Trim());
                cmd.Parameters.AddWithValue("@p_id_categoria", o.IdCategoriaFK.Trim());
                cmd.Parameters.AddWithValue("@p_id_ubicacion", o.IdUbicacionFK.Trim());
                cmd.Parameters.AddWithValue("@p_id_usuario", o.IdUsuarioFK.Trim());

                var outParam = new MySqlParameter("@p_nuevo_id_obra", MySqlDbType.String, 36)
                {
                    Direction = ParameterDirection.Output
                };
                cmd.Parameters.Add(outParam);

                cmd.ExecuteNonQuery();
                o.IdObra = outParam.Value?.ToString() ?? idEntrada;
                return true;
            }
            catch (Exception ex)
            {
                mensajeError = ex.Message;
                return false;

                conexion.Cerrar();

        // 4. Obras por ubicación (sector, barrio o comuna, uniendo UBICACION - BARRIO - COMUNA)
        public DataTable ConsultarObrasPorComunaOBarrio(string? idComuna, string? idBarrio)
        {
            using var conexion = new ConexionMysql();
            string sql = @"
                SELECT CONCAT('Comuna ', com.numeroComuna) AS 'Comuna',
                       b.nombreBarrio AS 'Barrio',
                       u.nombreLugar AS 'Sector / Lugar',
                       u.direccion AS 'Dirección',
                       o.nombreObra AS 'Obra de Arte Urbano',
                       a.nombreArtistico AS 'Artista',
                       c.nombreCategoria AS 'Categoría',
                       o.estadoConservacion AS 'Estado'
                FROM obra o
                INNER JOIN ubicacion u ON o.idUbicacionFK = u.idUbicacion
                INNER JOIN barrio b ON u.idBarrioFK = b.idBarrio
                INNER JOIN comuna com ON b.idComunaFK = com.idComuna
                INNER JOIN artista a ON o.idArtistaFK = a.idArtista
                INNER JOIN categoria c ON o.idCategoriaFK = c.idCategoria
                WHERE (@idComuna IS NULL OR com.idComuna = @idComuna)
                  AND (@idBarrio IS NULL OR b.idBarrio = @idBarrio)
                ORDER BY com.numeroComuna, b.nombreBarrio, o.nombreObra;";

            var p = new Dictionary<string, object?>
            {
                { "@idComuna", string.IsNullOrWhiteSpace(idComuna) ? null : idComuna.Trim() },
                { "@idBarrio", string.IsNullOrWhiteSpace(idBarrio) ? null : idBarrio.Trim() }
            };
            return conexion.ExecuteDataTable(sql, p);
        }

        // 5. Consulta de ruta: obras que la componen, orden de recorrido y ubicación de cada obra
        public DataTable ConsultarItinerarioRuta(string? idRuta = null)
        {
            using var conexion = new ConexionMysql();
            string sql = @"
                SELECT nombre_ruta AS 'Ruta Turística',
                       orden_visita AS 'Orden #',
                       titulo_obra AS 'Obra de Arte Urbano',
                       artista AS 'Artista',
                       nombreLugar AS 'Lugar / Sector',
                       direccion AS 'Dirección',
                       barrio AS 'Barrio',
                       comuna AS 'Comuna',
                       tiempo_permanencia_min AS 'Permanencia (min)',
                       recomendacion_parada AS 'Recomendación'
                FROM vw_rutas_itinerario_detallado ";

            Dictionary<string, object?>? p = null;
            if (!string.IsNullOrWhiteSpace(idRuta))
            {
                sql += "WHERE idRuta = @idRuta ";
                p = new Dictionary<string, object?> { { "@idRuta", idRuta.Trim() } };
            }

            sql += "ORDER BY nombre_ruta, orden_visita ASC;";
            return conexion.ExecuteDataTable(sql, p);
        }
